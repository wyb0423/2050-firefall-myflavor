# Country-specific definitions are loaded before generic government types so TUR receives distinct
# Ottoman monarchical and Turkish republican names without replacing the vanilla TUR country identity.

gov_ffpa_ottoman_autocracy = {
	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_SULTAN"
	female_ruler = "" # Preserve the vanilla Ottoman sultanate behavior.
	male_heir = "RULER_TITLE_IMPERIAL_PRINCE"
	female_heir = "" # Preserve the vanilla Ottoman sultanate behavior.

	possible = {
		c:TUR ?= this
		has_law_or_variant = law_type:law_monarchy
		has_law_or_variant = law_type:law_autocracy
		has_gov_regency = no
		is_domain_alliance_gov = no
	}

	on_government_type_change = { change_to_hereditary = yes }
	on_post_government_type_change = { post_change_to_hereditary = yes }
}
