# Country-specific definitions are loaded before generic government types so BYZ receives readable
# imperial and consular titles without changing any law or succession mechanics.

gov_ffpa_byzantine_autocracy = {
	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_FFPA_BASILEUS"
	female_ruler = "RULER_TITLE_FFPA_BASILISSA"
	male_heir = "RULER_TITLE_PRINCE"
	female_heir = "RULER_TITLE_PRINCESS"

	possible = {
		c:BYZ ?= this
		has_law_or_variant = law_type:law_monarchy
		has_law_or_variant = law_type:law_autocracy
		has_gov_regency = no
		is_domain_alliance_gov = no
	}

	on_government_type_change = { change_to_hereditary = yes }
	on_post_government_type_change = { post_change_to_hereditary = yes }
}
