gov_ffpa_usa_federal_government = {
	transfer_of_power = presidential_elective
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_PRESIDENT"
	female_ruler = "RULER_TITLE_PRESIDENT"

	possible = {
		country_definition = cd:USA
		has_law_or_variant = law_type:law_presidential_republic
		country_has_voting_franchise = yes
		NOR = {
			has_law_or_variant = law_type:law_single_party_state
			has_law_or_variant = law_type:law_technocracy
		}
		has_gov_regency = no
		is_domain_alliance_gov = no
	}

	on_government_type_change = { change_to_presidential_elective = yes }
	on_post_government_type_change = { post_change_to_presidential_elective = yes }
