# Keep the adapter journal available on new campaigns and existing saves.
# This must be a scripted effect: callers in ffpa_on_actions invoke it with
# `ffpa_ensure_auto_pm_compat_journal = yes`.
ffpa_ensure_auto_pm_compat_journal = {
	if = {
		limit = {
			is_player = yes
			NOT = { has_journal_entry = je_ffpa_auto_pm_compat }
		}
		add_journal_entry = { type = je_ffpa_auto_pm_compat }
	}
	if = {
		limit = {
			is_player = no
			NOT = { has_variable = ffpa_ai_building_pruning_initialized }
		}
		set_variable = ffpa_private_building_pruning_active
		set_variable = ffpa_government_building_pruning_active
		set_variable = ffpa_ai_building_pruning_initialized
	}
	if = {
		limit = {
			is_player = yes
			NOT = { has_journal_entry = je_ffpa_building_pruning }
		}
		add_journal_entry = { type = je_ffpa_building_pruning }
	}
}
