ffpa_ensure_tur_flavor_v1 = {
	if = {
		limit = { ROOT = { country_definition = cd:TUR } }
		ffpa_ensure_tur_frontier_recovery_v1 = yes

		# Infer only outcomes that leave an unambiguous persistent trace. The
		# expired pragmatic modifier cannot be reconstructed safely, so unresolved
		# saves receive a player-facing repair decision instead.
		if = {
			limit = { NOT = { has_variable = ffpa_tur_state_project_v1 } }
			if = {
				limit = {
					OR = {
						has_journal_entry = je_ffpa_tur_ottoman_commonwealth
						has_variable = ffpa_tur_ottoman_commonwealth_complete
						country_has_primary_culture = cu:ffpa_ottoman
						has_modifier = ffpa_ottoman_civic_compact
					}
				}
				ffpa_set_tur_state_project_ottoman_v1 = yes
			}
			else_if = {
				limit = { has_modifier = ffpa_tur_anatolian_citizenship }
				ffpa_set_tur_state_project_republic_v1 = yes
			}
			else_if = {
				limit = { has_modifier = ffpa_tur_reconstruction_pragmatism }
				ffpa_set_tur_state_project_directorate_v1 = yes
			}
			else_if = {
				limit = { has_variable = ffpa_tur_rebuild_anatolia_complete }
				set_variable = ffpa_tur_state_project_repair_needed_v1
			}
		}

		if = {
			limit = {
				country_has_primary_culture = cu:ffpa_ottoman
				NOT = { has_journal_entry = je_ffpa_tur_ottoman_commonwealth }
				NOT = { has_variable = ffpa_tur_ottoman_commonwealth_complete }
			}
			set_variable = ffpa_tur_ottoman_commonwealth_complete
		}

		if = {
			limit = { ffpa_tur_has_ottoman_state_project_v1 = yes }
			if = {
				limit = {
					NOT = { has_variable = ffpa_tur_ottoman_commonwealth_complete }
					NOT = { has_journal_entry = je_ffpa_tur_ottoman_commonwealth }
				}
				add_journal_entry = { type = je_ffpa_tur_ottoman_commonwealth }
			}
			else_if = {
				limit = {
					has_variable = ffpa_tur_ottoman_commonwealth_complete
					NOT = { has_variable = ffpa_tur_porte_charter_complete_v1 }
					NOT = { has_journal_entry = je_ffpa_tur_charter_of_the_porte }
				}
				add_journal_entry = { type = je_ffpa_tur_charter_of_the_porte }
			}
		}

		if = {
			limit = {
				ffpa_tur_has_republican_state_project_v1 = yes
				NOT = { has_variable = ffpa_tur_second_foundation_complete_v1 }
				NOT = { has_journal_entry = je_ffpa_tur_second_foundation_of_ankara }
			}
			add_journal_entry = { type = je_ffpa_tur_second_foundation_of_ankara }
		}

		if = {
			limit = {
				ffpa_tur_has_directorate_state_project_v1 = yes
				NOT = { has_variable = ffpa_tur_reconstruction_directorate_complete_v1 }
				NOT = { has_journal_entry = je_ffpa_tur_reconstruction_directorate }
			}
			add_journal_entry = { type = je_ffpa_tur_reconstruction_directorate }
		}

		ffpa_ensure_tur_route_event_chains_v1 = yes

		if = {
			limit = {
				ffpa_tur_has_ottoman_state_project_v1 = yes
				has_variable = ffpa_tur_porte_charter_complete_v1
				NOT = { has_variable = ffpa_tur_porte_charter_choice_v1 }
				NOT = { has_variable = ffpa_tur_porte_charter_capstone_scheduled_v1 }
			}
			set_variable = ffpa_tur_porte_charter_capstone_scheduled_v1
			trigger_event = { id = ffpa_tur_flavor.13 days = 1 }
		}
		if = {
			limit = {
				ffpa_tur_has_republican_state_project_v1 = yes
				has_variable = ffpa_tur_second_foundation_complete_v1
				NOT = { has_variable = ffpa_tur_republic_settlement_choice_v1 }
				NOT = { has_variable = ffpa_tur_republic_capstone_scheduled_v1 }
			}
			set_variable = ffpa_tur_republic_capstone_scheduled_v1
			trigger_event = { id = ffpa_tur_flavor.23 days = 1 }
		}
		if = {
			limit = {
				ffpa_tur_has_directorate_state_project_v1 = yes
				has_variable = ffpa_tur_reconstruction_directorate_complete_v1
				NOT = { has_variable = ffpa_tur_directorate_settlement_choice_v1 }
				NOT = { has_variable = ffpa_tur_directorate_capstone_scheduled_v1 }
			}
			set_variable = ffpa_tur_directorate_capstone_scheduled_v1
			trigger_event = { id = ffpa_tur_flavor.33 days = 1 }
		}
		if = {
			limit = {
				has_variable = ffpa_tur_new_straits_convention_complete_v1
				NOT = { has_variable = ffpa_tur_straits_convention_choice_v1 }
				NOT = { has_variable = ffpa_tur_straits_convention_event_scheduled_v1 }
			}
			set_variable = ffpa_tur_straits_convention_event_scheduled_v1
			trigger_event = { id = ffpa_tur_flavor.41 days = 1 }
		}

		ffpa_ensure_tur_permanent_flavor_rewards_v1 = yes
		ffpa_check_tur_post_reconstruction_events_v1 = yes
	}
}
