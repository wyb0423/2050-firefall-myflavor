ffpa_ensure_tur_flavor_v1 = {
	if = {
		limit = { c:TUR ?= ROOT }

		# Infer only outcomes that leave an unambiguous persistent trace. The
		# expired pragmatic modifier cannot be reconstructed safely, so unresolved
		# saves receive a player-facing repair decision instead.
		if = {
			limit = { NOT = { has_variable = ffpa_tur_state_project_v1 } }
			if = {
				limit = {
					OR = {
						has_journal_entry = je_ffpa_tur_ottoman_commonwealth
						has_variable = ffpa_tur_ottoman_commonwealth_complete
						country_has_primary_culture = cu:ffpa_ottoman
						has_modifier = ffpa_ottoman_civic_compact
					}
				}
				ffpa_set_tur_state_project_ottoman_v1 = yes
			}
			else_if = {
				limit = { has_modifier = ffpa_tur_anatolian_citizenship }
				ffpa_set_tur_state_project_republic_v1 = yes
			}
			else_if = {
				limit = { has_modifier = ffpa_tur_reconstruction_pragmatism }
				ffpa_set_tur_state_project_directorate_v1 = yes
			}
			else_if = {
				limit = { has_variable = ffpa_tur_rebuild_anatolia_complete }
				set_variable = ffpa_tur_state_project_repair_needed_v1
			}
		}

		if = {
			limit = {
				country_has_primary_culture = cu:ffpa_ottoman
				NOT = { has_journal_entry = je_ffpa_tur_ottoman_commonwealth }
				NOT = { has_variable = ffpa_tur_ottoman_commonwealth_complete }
			}
			set_variable = ffpa_tur_ottoman_commonwealth_complete
		}

		if = {
			limit = { ffpa_tur_has_ottoman_state_project_v1 = yes }
			if = {
				limit = {
					NOT = { has_variable = ffpa_tur_ottoman_commonwealth_complete }
					NOT = { has_journal_entry = je_ffpa_tur_ottoman_commonwealth }
				}
				add_journal_entry = { type = je_ffpa_tur_ottoman_commonwealth }
			}
			else_if = {
				limit = {
					has_variable = ffpa_tur_ottoman_commonwealth_complete
					NOT = { has_variable = ffpa_tur_porte_charter_complete_v1 }
					NOT = { has_journal_entry = je_ffpa_tur_charter_of_the_porte }
				}
				add_journal_entry = { type = je_ffpa_tur_charter_of_the_porte }
			}
