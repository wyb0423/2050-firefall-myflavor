# Country scope. Persistent per-province flags survive target state merges.
# Pass 1 snapshots, 2 counts, 3 clears, 4 rewards each receiving state once.
ffpa_na_union_visit_province_v1 = {
 if = {
  limit = { var:ffpa_na_union_pass_v1 = 1 }
  if = {
   limit = { p:$PROVINCE$.state = root.var:ffpa_na_union_target_v1 }
   set_variable = ffpa_na_union_p_$PROVINCE$_v1
   change_variable = { name = ffpa_na_union_total_v1 add = 1 }
  }
 }
 else_if = {
  limit = { var:ffpa_na_union_pass_v1 = 2 has_variable = ffpa_na_union_p_$PROVINCE$_v1 }
  if = {
   limit = { exists = p:$PROVINCE$.state.owner }
   change_variable = { name = ffpa_na_union_seen_v1 add = 1 }
   if = {
    limit = { p:$PROVINCE$.state.owner = root }
    change_variable = { name = ffpa_na_union_owned_v1 add = 1 }
   }
  }
 }
 else_if = {
  limit = { var:ffpa_na_union_pass_v1 = 3 }
  remove_variable = ffpa_na_union_p_$PROVINCE$_v1
 }
 else_if = {
  limit = { var:ffpa_na_union_pass_v1 = 4 has_variable = ffpa_na_union_p_$PROVINCE$_v1 p:$PROVINCE$.state.owner = root }
  # Country -> receiving state; the temporary marker prevents one award per province.
  p:$PROVINCE$.state = {
   if = {
    limit = { NOT = { has_variable = ffpa_na_union_awarding_v1 } }
    set_variable = ffpa_na_union_awarding_v1
    if = { limit = { has_modifier = ffpa_na_union_market_v1 } remove_modifier = ffpa_na_union_market_v1 }
    if = { limit = { has_modifier = ffpa_na_union_government_v1 } remove_modifier = ffpa_na_union_government_v1 }
    if = {
     limit = { root.var:ffpa_na_union_choice_v1 = 1 }
     add_modifier = { name = ffpa_na_union_market_v1 years = 5 }
    }
    else = { add_modifier = { name = ffpa_na_union_government_v1 years = 5 } }
   }
  }
 }
}

  limit = { ffpa_na_union_can_start_v1 = yes }

  set_variable = { name = ffpa_na_union_region_v1 value = scope:ffpa_na_union_selected_region }

  set_variable = { name = ffpa_na_union_pass_v1 value = 1 }

  remove_variable = ffpa_na_union_pass_v1

   limit = { var:ffpa_na_union_total_v1 > 0 }
   change_variable = { name = ffpa_na_union_serial_v1 add = 1 }
   set_variable = { name = ffpa_na_union_status_v1 value = 1 }

    limit = { NOT = { has_claim = var:ffpa_na_union_region_v1 } }
    var:ffpa_na_union_region_v1 = { add_claim = root }
    # Set AFTER native add_claim; delayed callbacks conservatively preserve the claim.
    set_variable = ffpa_na_union_claim_v1
   }
   ffpa_na_union_recount_v1 = yes
  }
  else = { ffpa_na_union_cleanup_v1 = yes }
 }
}

    ffpa_na_union_fail_v1 = yes

    trigger_event = { id = ffpa_na_flavor.4 days = 1 }

   else_if = {
    limit = { NOT = { has_variable = ffpa_na_union_deadline_v1 } }
    ffpa_na_union_fail_v1 = yes
   }
  }
 }
 else_if = {
  limit = { ffpa_na_union_idle_v1 = yes is_ai = yes NOT = { has_variable = ffpa_na_union_ai_refresh_v1 } }
