# Cold War Era uses one state/removal statement per building type because
# remove_building is a state effect and requires a concrete building type.
# Parameterizing that repeated statement keeps the compatibility list auditable.
ffpa_prune_private_building_type = {
	every_scope_state = {
		if = {
			limit = {
				any_scope_building = {
					is_building_type = $BUILDING$
					ffpa_is_prunable_private_building = yes
				}
			}
			ROOT = { set_variable = ffpa_private_building_pruned_this_run }
			remove_building = $BUILDING$
		}
	}
}
