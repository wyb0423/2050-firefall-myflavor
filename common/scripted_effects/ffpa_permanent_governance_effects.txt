	if = {
		limit = { var:ffpa_tur_capital_anomalies_v1 = 0 }
		change_variable = { name = ffpa_tur_capital_reserve_v1 add = 3 }
	}
	else_if = {
		limit = { var:ffpa_tur_capital_anomalies_v1 = 1 }
		change_variable = { name = ffpa_tur_capital_reserve_v1 add = 1 }
	}
	else_if = {
		limit = { var:ffpa_tur_capital_anomalies_v1 = 2 }
		change_variable = { name = ffpa_tur_capital_reserve_v1 subtract = 2 }
	}
	else_if = {
		limit = { var:ffpa_tur_capital_anomalies_v1 = 3 }
		change_variable = { name = ffpa_tur_capital_reserve_v1 subtract = 5 }
	}
	else = { change_variable = { name = ffpa_tur_capital_reserve_v1 subtract = 8 } }
	if = {
		limit = { ffpa_tur_capital_has_famine_v1 = yes }
		change_variable = { name = ffpa_tur_capital_reserve_v1 subtract = 10 }
	}
	if = {
		limit = { NOT = { ffpa_tur_capital_supply_severe_crisis_v1 = yes } }
		remove_variable = ffpa_tur_capital_emergency_used_v1
	}
	clamp_variable = { name = ffpa_tur_capital_reserve_v1 min = 0 max = 100 }
	if = {
		limit = { var:ffpa_tur_capital_reserve_v1 > 0 }
		ffpa_reset_tur_capital_supply_failure_counters_v1 = yes
	}
	else = {
		if = {
			limit = { ffpa_tur_capital_supply_severe_crisis_v1 = yes }
			change_variable = { name = ffpa_tur_capital_crisis_months_v1 add = 1 }
		}
		else = { set_variable = { name = ffpa_tur_capital_crisis_months_v1 value = 0 } }

			limit = { ffpa_tur_capital_has_famine_v1 = yes }
			change_variable = { name = ffpa_tur_capital_famine_months_v1 add = 1 }
		}
		else = { set_variable = { name = ffpa_tur_capital_famine_months_v1 value = 0 } }
	}
	clamp_variable = { name = ffpa_tur_capital_crisis_months_v1 min = 0 max = 6 }
	clamp_variable = { name = ffpa_tur_capital_famine_months_v1 min = 0 max = 3 }
	ffpa_refresh_tur_supply_tier_modifier_v1 = yes
	ffpa_sync_permanent_governance_progress_bars_v1 = yes
