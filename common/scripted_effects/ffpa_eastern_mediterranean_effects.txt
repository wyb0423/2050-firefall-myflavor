# Country scope. These packages replace only a known vanilla or Tech & Res
# ideology. If another mod has already replaced that slot, its result wins and
# the adapter records the package as handled without appending another one.
ffpa_apply_tur_formation_ideologies_v1 = {
	if = {
		limit = {
			ROOT = { country_definition = cd:TUR }
			NOT = { has_variable = ffpa_tur_formation_ideologies_v1 }
		}

		ig:ig_armed_forces = {
			if = {
				limit = { has_ideology = ideology:ideology_jingoist }
				remove_ideology = ideology_jingoist
				if = {
					limit = { NOT = { has_ideology = ideology:ideology_ffpa_anatolian_reconstructionism } }
					add_ideology = ideology_ffpa_anatolian_reconstructionism
				}
			}
		}

		ig:ig_industrialists = {
			if = {
				limit = {
					OR = {
						has_ideology = ideology:ideology_laissez_faire
						has_ideology = ideology:ideology_neoliberism
					}
				}
				if = {
					limit = { has_ideology = ideology:ideology_laissez_faire }
					remove_ideology = ideology_laissez_faire
				}
				if = {
					limit = { has_ideology = ideology:ideology_neoliberism }
					remove_ideology = ideology_neoliberism
				}
				if = {
					limit = { NOT = { has_ideology = ideology:ideology_ffpa_tur_state_developmentalism } }
					add_ideology = ideology_ffpa_tur_state_developmentalism
				}
			}
		}

		ig:ig_petty_bourgeoisie = {
			if = {
				limit = { has_ideology = ideology:ideology_reactionary }
				remove_ideology = ideology_reactionary
				if = {
					limit = { NOT = { has_ideology = ideology:ideology_ffpa_anatolian_civic_statism } }
					add_ideology = ideology_ffpa_anatolian_civic_statism
				}
			}
		}

		set_variable = ffpa_tur_formation_ideologies_v1
	}
}

# Root = BYZ. Replace the original permanent reward without stacking its
# unchanged acceptance, loyalism, migration, and qualification modifiers.
# The completed variable, Rhomaic primary culture, or legacy modifier can each
# recover old saves whose completion event has already fired.
ffpa_ensure_byz_rhomaic_civic_compact_v2 = {
	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }
			OR = {
				has_variable = ffpa_byz_rhomaic_commonwealth_complete
				country_has_primary_culture = cu:ffpa_rhomaic
				has_modifier = ffpa_rhomaic_civic_compact
			}
		}
		if = {
			limit = { has_modifier = ffpa_rhomaic_civic_compact }
			remove_modifier = ffpa_rhomaic_civic_compact
		}
		if = {
			limit = { NOT = { has_modifier = ffpa_rhomaic_civic_compact_v2 } }
			add_modifier = { name = ffpa_rhomaic_civic_compact_v2 }
		}
	}
}

# Root = GRE. Firefall successor countries can form Greece while retaining a
# Sunni runtime state religion. Normalize the Greek identity once, without
# overriding State Atheism; leaving State Atheism later restores the Orthodox
# religion declared by GRE's country definition.
ffpa_ensure_greek_formation_religion_v1 = {
	if = {
		limit = {
			ROOT = { country_definition = cd:GRE }
			NOT = { has_variable = ffpa_greek_state_religion_normalized_v1 }
		}

		if = {
			limit = {
				NOT = { has_law_or_variant = law_type:law_state_atheism }
				NOT = { country_has_state_religion = rel:orthodox }
			}
			set_state_religion = rel:orthodox
			state_religion_switch_effect = yes
		}

		set_variable = ffpa_greek_state_religion_normalized_v1
	}
}

# Root = BYZ. One-shot package for formation and old-save repair. Each
# set_ig_trait call replaces only the trait in the matching approval slot.
ffpa_ensure_byzantine_interest_group_traits_v1 = {
	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }
			NOT = { has_variable = ffpa_byzantine_interest_group_traits_v1 }
		}

		ig:ig_devout ?= {
			set_ig_trait = ig_trait:ig_trait_ffpa_byz_synodal_deadlock
			set_ig_trait = ig_trait:ig_trait_ffpa_byz_synodal_stewardship
		}
		ig:ig_armed_forces ?= {
			set_ig_trait = ig_trait:ig_trait_ffpa_byz_strategikon
			set_ig_trait = ig_trait:ig_trait_ffpa_byz_restored_eagles
		}
		ig:ig_intelligentsia ?= {
			set_ig_trait = ig_trait:ig_trait_ffpa_byz_new_rome_academies
		}

		set_variable = ffpa_byzantine_interest_group_traits_v1
	}

	# Orthodox is the vanilla fallback, but changing state religion does not
	# refresh IG traits; clear Da'wat inherited from a Sunni predecessor.
	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }
			NOT = { has_variable = ffpa_byzantine_devout_loyal_trait_v1 }
		}
		ig:ig_devout ?= {
			set_ig_trait = ig_trait:ig_trait_be_fruitful_and_multiply
		}
		set_variable = ffpa_byzantine_devout_loyal_trait_v1
	}
}

# Root = BYZ. Safe for both formation-time execution and old-save repair.
ffpa_ensure_byzantine_formation_effects = {
	# BYZ's static country definition does not replace the runtime religion
	# inherited through GRE. Keep this migration separate from the older
	# formation-effects key so existing BYZ saves receive the repair once.
	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }
			NOT = { has_variable = ffpa_byzantine_state_religion_normalized_v1 }
		}

		if = {
			limit = {
				NOT = { has_law_or_variant = law_type:law_state_atheism }
				NOT = { country_has_state_religion = rel:orthodox }
			}
			set_state_religion = rel:orthodox
			state_religion_switch_effect = yes
		}

		set_variable = ffpa_byzantine_state_religion_normalized_v1
	}

	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }
			NOT = { has_variable = ffpa_byzantine_formation_incorporation_v2 }
		}

		every_owned_state = {
			limit = {
				is_incorporated = no
				OR = {
					state_region = s:STATE_EASTERN_THRACE
					state_region = s:STATE_ATTICA
					state_region = s:STATE_THESSALIA
					state_region = s:STATE_MACEDONIA
					state_region = s:STATE_SKOPIA
					state_region = s:STATE_WESTERN_THRACE
					state_region = s:STATE_NORTHERN_THRACE
					state_region = s:STATE_HUDAVENDIGAR
					state_region = s:STATE_AYDIN
					state_region = s:STATE_WEST_AEGEAN_ISLANDS
					state_region = s:STATE_EAST_AEGEAN_ISLANDS
					state_region = s:STATE_ALBANIA
				}
			}
			set_state_type = incorporated
		}

		set_variable = ffpa_byzantine_formation_effects_v1
		set_variable = ffpa_byzantine_formation_incorporation_v2
	}

	# v2 supersedes the former supplemental-only marker. Existing claims are
	# preserved; this migration only grants missing states from the new whitelist.
	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }
			NOT = { has_variable = ffpa_byzantine_justinianic_claims_v2 }
		}
		ffpa_grant_byzantine_justinianic_claims_v2 = yes
		set_variable = ffpa_byzantine_justinianic_claims_v1
		set_variable = ffpa_byzantine_justinianic_claims_v2
	}

	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }
			NOT = { has_variable = ffpa_byzantine_flavor_names_v2 }
		}
		ig:ig_devout = { set_interest_group_name = ffpa_ig_byz_holy_synod }
		ig:ig_armed_forces = { set_interest_group_name = ffpa_ig_byz_roman_army }
		ig:ig_intelligentsia = { set_interest_group_name = ffpa_ig_byz_new_rome_society }
		set_variable = ffpa_byzantine_flavor_names_v2
	}

	ffpa_ensure_byzantine_interest_group_traits_v1 = yes
}

# Root = state. Dynamic incorporation support is recalculated from the current
# owner, the stable western whitelist, and the authoritative v2 route choice.
# Clearing all three variants first keeps repeated on_action calls idempotent.
ffpa_refresh_byz_western_citizenship_incorporation_state_v1 = {
	if = {
		limit = { has_modifier = ffpa_byz_western_universal_incorporation_state_v1 }
		remove_modifier = ffpa_byz_western_universal_incorporation_state_v1
	}
	if = {
		limit = { has_modifier = ffpa_byz_western_provincial_incorporation_state_v1 }
		remove_modifier = ffpa_byz_western_provincial_incorporation_state_v1
	}
	if = {
		limit = { has_modifier = ffpa_byz_western_service_incorporation_state_v1 }
		remove_modifier = ffpa_byz_western_service_incorporation_state_v1
	}

	if = {
		limit = {
			ffpa_is_byzantine_western_integration_state = yes
			is_incorporated = no
			# State -> current owner country.
			owner ?= {
				country_definition = cd:BYZ
				has_variable = ffpa_byz_western_citizenship_choice_v2
				var:ffpa_byz_western_citizenship_choice_v2 = 1
			}
		}
		add_modifier = { name = ffpa_byz_western_universal_incorporation_state_v1 }
	}
	else_if = {
		limit = {
			ffpa_is_byzantine_western_integration_state = yes
			is_incorporated = no
			# State -> current owner country.
			owner ?= {
				country_definition = cd:BYZ
				has_variable = ffpa_byz_western_citizenship_choice_v2
				var:ffpa_byz_western_citizenship_choice_v2 = 2
			}
		}
		add_modifier = { name = ffpa_byz_western_provincial_incorporation_state_v1 }
	}
	else_if = {

# Root = BYZ. Old `.30`, `.31`, and `.33` choices cannot be reconstructed once
# their finite modifiers expire. Reviews are therefore serialized through the
# original event IDs. `.32` is included only as a fallback for damaged saves in
# which neither permanent v1 citizenship modifier remains.
ffpa_schedule_next_western_integration_reward_review_v2 = {
	if = {
		limit = {
			ROOT = { country_definition = cd:BYZ }

		}
		if = {

				has_variable = ffpa_byz_mediterranean_congress_fired_v1

			set_variable = ffpa_byz_mediterranean_congress_scheduled_v1

			limit = {
				has_variable = ffpa_byz_western_cadastre_fired_v1
				NOT = { has_variable = ffpa_byz_western_cadastre_choice_v2 }
			}
			set_variable = ffpa_byz_western_cadastre_review_v2
			set_variable = ffpa_byz_western_reward_review_active_v2
			set_variable = ffpa_byz_western_cadastre_scheduled_v1
			trigger_event = { id = ffpa_flavor.31 days = 7 }
		}
		else_if = {
			limit = {
				has_variable = ffpa_byz_western_civic_extension_fired_v1
				NOT = { has_variable = ffpa_byz_western_citizenship_choice_v2 }
			}
			set_variable = ffpa_byz_western_civic_extension_review_v2
			set_variable = ffpa_byz_western_reward_review_active_v2
			set_variable = ffpa_byz_western_civic_extension_scheduled_v1
			trigger_event = { id = ffpa_flavor.32 days = 7 }
		}

			limit = {
				has_variable = ffpa_byz_western_representation_fired_v1
				NOT = { has_variable = ffpa_byz_western_representation_choice_v2 }
			}

			set_variable = ffpa_byz_western_reward_review_active_v2
			set_variable = ffpa_byz_western_representation_scheduled_v1
