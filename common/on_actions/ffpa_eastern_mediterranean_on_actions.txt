on_country_formed = {
	on_actions = {
		ffpa_on_eastern_mediterranean_country_formed
	}
}

ffpa_on_eastern_mediterranean_country_formed = {
	effect = {
		ffpa_handle_tur_flavor_country_formed_v1 = yes
		ffpa_handle_permanent_governance_country_formed_v1 = yes
		if = {
			limit = { ROOT = { country_definition = cd:TUR } }
			ffpa_apply_tur_formation_ideologies_v1 = yes
		}
		if = {
			limit = { ROOT = { country_definition = cd:GRE } }
			ffpa_ensure_greek_formation_religion_v1 = yes
		}
		if = {
			limit = { ROOT = { country_definition = cd:BYZ } }
			ffpa_ensure_byzantine_formation_effects = yes
			ffpa_apply_byz_formation_ideologies_v1 = yes
		}
		if = {
			limit = {
				ROOT = { NOT = { country_definition = cd:BYZ } }
				OR = {
					has_journal_entry = je_ffpa_byz_restore_balkans
					has_journal_entry = je_ffpa_byz_restore_anatolia
					has_journal_entry = je_ffpa_byz_restore_oriens
					has_journal_entry = je_ffpa_byz_restore_aegyptus
					has_journal_entry = je_ffpa_byz_restore_africa
					has_journal_entry = je_ffpa_byz_restore_italia
					has_journal_entry = je_ffpa_byz_restore_spania
					has_variable = ffpa_byz_balkans_restored_complete_v1
					has_variable = ffpa_byz_anatolia_restored_complete_v1
					has_variable = ffpa_byz_oriens_restored_complete_v1
					has_variable = ffpa_byz_aegyptus_restored_complete_v1
					has_variable = ffpa_byz_africa_restored_complete_v1
					has_variable = ffpa_byz_italia_restored_complete_v1
					has_variable = ffpa_byz_spania_restored_complete_v1
					has_variable = ffpa_byz_balkans_completion_predecessor_ready_v1
					has_variable = ffpa_byz_anatolia_completion_predecessor_ready_v1
					has_variable = ffpa_byz_oriens_completion_predecessor_ready_v1
					has_variable = ffpa_byz_aegyptus_completion_predecessor_ready_v1
					has_variable = ffpa_byz_africa_completion_predecessor_ready_v1
					has_variable = ffpa_byz_italia_completion_predecessor_ready_v1
					has_variable = ffpa_byz_spania_completion_predecessor_ready_v1
				}
			}
			ffpa_cleanup_all_byz_restoration_predecessor_tracking_v1 = yes
		}
		if = {
			limit = {
				OR = {
					ROOT = { country_definition = cd:BYZ }
					any_scope_state = {
						OR = {
							has_modifier = ffpa_byz_western_universal_incorporation_state_v1
							has_modifier = ffpa_byz_western_provincial_incorporation_state_v1
							has_modifier = ffpa_byz_western_service_incorporation_state_v1
						}
					}
				}
			}
			# Country -> owned states. This also clears rewards if BYZ forms another tag.
			ffpa_refresh_byz_western_citizenship_incorporation_states_v1 = yes
		}
	}
}

on_state_incorporation = {

	}
}

on_state_owner_change = {

# Root = state. Ordinary ownership changes stop at the shallow tracking-marker
# guard; only a state currently watched by one of the fifteen campaigns enters
# the country-specific snapshot handlers.
ffpa_on_eastern_mediterranean_predecessor_owner_change_v1 = {
	trigger = {
		OR = {
			has_variable = ffpa_tur_front_western_pact_owner_tracking_v1
			has_variable = ffpa_tur_front_mosul_owner_tracking_v1
			has_variable = ffpa_tur_front_rumelia_owner_tracking_v1
			has_variable = ffpa_tur_front_aegean_cyprus_owner_tracking_v1
			has_variable = ffpa_tur_front_levant_owner_tracking_v1
			has_variable = ffpa_tur_front_mesopotamia_owner_tracking_v1
			has_variable = ffpa_tur_front_egypt_owner_tracking_v1
			has_variable = ffpa_tur_front_ifriqiya_owner_tracking_v1
			has_variable = ffpa_byz_balkans_owner_tracking_v1
			has_variable = ffpa_byz_anatolia_owner_tracking_v1
			has_variable = ffpa_byz_oriens_owner_tracking_v1
			has_variable = ffpa_byz_aegyptus_owner_tracking_v1
			has_variable = ffpa_byz_africa_owner_tracking_v1
			has_variable = ffpa_byz_italia_owner_tracking_v1
			has_variable = ffpa_byz_spania_owner_tracking_v1
		}
	}
	effect = {
		ffpa_track_tur_restoration_predecessors_on_owner_change_state_v1 = yes
		ffpa_track_byz_restoration_predecessors_on_owner_change_state_v1 = yes
	}
}

# Root = newly created state. State creation is rare; each country handler
# directly checks the unique TUR/BYZ tag, its active JE, and the new state's
# region before adding a campaign-specific snapshot.
ffpa_on_eastern_mediterranean_predecessor_state_created_v1 = {
	effect = {
		ffpa_seed_tur_restoration_predecessors_on_state_created_v1 = yes
		ffpa_seed_byz_restoration_predecessors_on_state_created_v1 = yes
	}
}

# Triggered only by state-scoped code on_actions. Existing dynamic modifiers
# always enter so ownership loss, incorporation, or later whitelist changes can
# clean them. Eligible new BYZ states enter through the second branch.
ffpa_on_byz_western_citizenship_state_changed_v1 = {
	trigger = {
		OR = {
			has_modifier = ffpa_byz_western_universal_incorporation_state_v1
			has_modifier = ffpa_byz_western_provincial_incorporation_state_v1
			has_modifier = ffpa_byz_western_service_incorporation_state_v1
			AND = {
				ffpa_is_byzantine_western_integration_state = yes
				is_incorporated = no
				# State -> current owner country.
				owner ?= {
					country_definition = cd:BYZ
					has_variable = ffpa_byz_western_citizenship_choice_v2
					OR = {
						var:ffpa_byz_western_citizenship_choice_v2 = 1
						var:ffpa_byz_western_citizenship_choice_v2 = 2
						var:ffpa_byz_western_citizenship_choice_v2 = 3
					}
				}
			}
		}
	}
	effect = {
		ffpa_refresh_byz_western_citizenship_incorporation_state_v1 = yes
	}
}

# One guarded country check provides versioned migration for GRE, TUR, and BYZ saves
# created before this adapter version. It performs no repeated ideology write,
# world traversal, or building traversal.
on_monthly_pulse_country = {
	on_actions = {
		ffpa_monthly_ensure_byzantine_formation_effects
	}
}

ffpa_monthly_ensure_byzantine_formation_effects = {
	effect = {
		if = {
			limit = {
				OR = {
					ROOT = { country_definition = cd:BYZ }
					ROOT = { country_definition = cd:GRE }
					ROOT = { country_definition = cd:TUR }
				}
			}
			ffpa_ensure_eastern_mediterranean_v2 = yes
		}
	}
}

on_election_campaign_start = {
	on_actions = {
		ffpa_on_eastern_mediterranean_first_election
	}
}

ffpa_on_eastern_mediterranean_first_election = {
	effect = {
		if = {
			limit = {
				ROOT = { country_definition = cd:TUR }
				has_variable = ffpa_tur_rebuild_anatolia_complete
				NOT = { has_variable = ffpa_tur_first_imperial_election_fired_v2 }
				NOT = { has_variable = ffpa_tur_first_imperial_election_scheduled_v3 }
			}
			set_variable = ffpa_tur_first_imperial_election_scheduled_v3
			trigger_event = { id = ffpa_flavor.12 days = 7 }
		}

		if = {
			limit = {
				ROOT = { country_definition = cd:BYZ }
				has_variable = ffpa_byz_new_rome_complete
				NOT = { has_variable = ffpa_byz_first_election_fired_v2 }
				NOT = { has_variable = ffpa_byz_first_election_scheduled_v3 }
			}
			set_variable = ffpa_byz_first_election_scheduled_v3
			trigger_event = { id = ffpa_flavor.22 days = 7 }
		}

		if = {
			limit = {
				ROOT = { country_definition = cd:BYZ }
				country_has_voting_franchise = yes
				ffpa_has_two_western_development_projects = yes
				NOT = { has_variable = ffpa_byz_western_representation_scheduled_v1 }
				NOT = { has_variable = ffpa_byz_western_representation_fired_v1 }
			}
			set_variable = ffpa_byz_western_representation_scheduled_v1
			trigger_event = { id = ffpa_flavor.33 days = 21 }
		}
	}
}
