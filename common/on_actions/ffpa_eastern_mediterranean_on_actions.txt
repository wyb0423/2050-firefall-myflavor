on_country_formed = {
	on_actions = {
		ffpa_on_eastern_mediterranean_country_formed
	}
}

ffpa_on_eastern_mediterranean_country_formed = {
	effect = {
		ffpa_handle_tur_flavor_country_formed_v1 = yes
		ffpa_handle_permanent_governance_country_formed_v1 = yes
		if = {
			limit = { ROOT = { country_definition = cd:TUR } }
			ffpa_apply_tur_formation_ideologies_v1 = yes
		}
		if = {
			limit = { ROOT = { country_definition = cd:GRE } }
			ffpa_ensure_greek_formation_religion_v1 = yes
		}
		if = {
			limit = { ROOT = { country_definition = cd:BYZ } }
			ffpa_ensure_byzantine_formation_effects = yes
			ffpa_apply_byz_formation_ideologies_v1 = yes
		}
		if = {
			limit = {
				OR = {
					ROOT = { country_definition = cd:BYZ }
					any_scope_state = {
						OR = {
							has_modifier = ffpa_byz_western_universal_incorporation_state_v1
							has_modifier = ffpa_byz_western_provincial_incorporation_state_v1
							has_modifier = ffpa_byz_western_service_incorporation_state_v1
						}
					}
				}
			}
			# Country -> owned states. This also clears rewards if BYZ forms another tag.
			ffpa_refresh_byz_western_citizenship_incorporation_states_v1 = yes
		}
	}
}

on_state_incorporation = {
	on_actions = {
		ffpa_on_byz_western_citizenship_state_changed_v1
	}
}

on_state_owner_change = {
	on_actions = {
		ffpa_on_byz_western_citizenship_state_changed_v1
	}
}

on_state_created = {
	on_actions = {
		ffpa_on_byz_western_citizenship_state_changed_v1
	}
}

# Triggered only by state-scoped code on_actions. Existing dynamic modifiers
# always enter so ownership loss, incorporation, or later whitelist changes can
# clean them. Eligible new BYZ states enter through the second branch.
ffpa_on_byz_western_citizenship_state_changed_v1 = {
	trigger = {
		OR = {
			has_modifier = ffpa_byz_western_universal_incorporation_state_v1
			has_modifier = ffpa_byz_western_provincial_incorporation_state_v1
			has_modifier = ffpa_byz_western_service_incorporation_state_v1
			AND = {
				ffpa_is_byzantine_western_integration_state = yes
				is_incorporated = no
				# State -> current owner country.
				owner ?= {
					country_definition = cd:BYZ
					has_variable = ffpa_byz_western_citizenship_choice_v2
					OR = {
						var:ffpa_byz_western_citizenship_choice_v2 = 1
						var:ffpa_byz_western_citizenship_choice_v2 = 2
						var:ffpa_byz_western_citizenship_choice_v2 = 3
					}
				}
			}
		}
	}
	effect = {
		ffpa_refresh_byz_western_citizenship_incorporation_state_v1 = yes
	}
}

# One guarded country check provides versioned migration for GRE, TUR, and BYZ saves
# created before this adapter version. It performs no repeated ideology write,
# world traversal, or building traversal.
on_monthly_pulse_country = {
	on_actions = {
		ffpa_monthly_ensure_byzantine_formation_effects
	}
}

ffpa_monthly_ensure_byzantine_formation_effects = {
	effect = {
		if = {
			limit = {
				OR = {
					ROOT = { country_definition = cd:BYZ }
					ROOT = { country_definition = cd:GRE }
					ROOT = { country_definition = cd:TUR }
				}
			}
			ffpa_ensure_eastern_mediterranean_v2 = yes
		}
	}
}

on_election_campaign_start = {
	on_actions = {
		ffpa_on_eastern_mediterranean_first_election
	}
}

ffpa_on_eastern_mediterranean_first_election = {
	effect = {
		if = {
			limit = {
				ROOT = { country_definition = cd:TUR }
				has_variable = ffpa_tur_rebuild_anatolia_complete
				NOT = { has_variable = ffpa_tur_first_imperial_election_fired_v2 }
				NOT = { has_variable = ffpa_tur_first_imperial_election_scheduled_v3 }
			}
			set_variable = ffpa_tur_first_imperial_election_scheduled_v3
			trigger_event = { id = ffpa_flavor.12 days = 7 }
		}

		if = {
			limit = {
				ROOT = { country_definition = cd:BYZ }
				has_variable = ffpa_byz_new_rome_complete
				NOT = { has_variable = ffpa_byz_first_election_fired_v2 }
				NOT = { has_variable = ffpa_byz_first_election_scheduled_v3 }
			}
			set_variable = ffpa_byz_first_election_scheduled_v3
			trigger_event = { id = ffpa_flavor.22 days = 7 }
		}

		if = {
			limit = {
				ROOT = { country_definition = cd:BYZ }
				country_has_voting_franchise = yes
				ffpa_has_two_western_development_projects = yes
				NOT = { has_variable = ffpa_byz_western_representation_scheduled_v1 }
				NOT = { has_variable = ffpa_byz_western_representation_fired_v1 }
			}
			set_variable = ffpa_byz_western_representation_scheduled_v1
			trigger_event = { id = ffpa_flavor.33 days = 21 }
		}
	}
}
