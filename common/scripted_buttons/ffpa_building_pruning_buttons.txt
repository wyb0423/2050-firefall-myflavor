ffpa_toggle_private_building_pruning_button = {
	name = "ffpa_toggle_private_building_pruning_button"
	desc = "ffpa_toggle_private_building_pruning_button_desc"
	selected = { has_variable = ffpa_private_building_pruning_active }
	possible = { is_at_war = no }

	effect = {
		if = {
			limit = { has_variable = ffpa_private_building_pruning_active }
			remove_variable = ffpa_private_building_pruning_active
		}
		else = {
			set_variable = ffpa_private_building_pruning_active
			ffpa_prune_private_buildings = yes
		}
	}
}
