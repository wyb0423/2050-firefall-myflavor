# Country variables remain the authoritative save-state for both commonwealth
# journals. Native monthly_progress supplies the standard/CMF UI breakdown;
# the journal pulse then mirrors the country value back into each bar.
ffpa_tur_commonwealth_administration_bar = {
	name = "ffpa_commonwealth_administration_bar"
	desc = "ffpa_commonwealth_administration_bar"
	second_desc = "ffpa_commonwealth_progress_value"
	monthly_progress = {
		if = {
			limit = { ffpa_tur_commonwealth_administration_conditions = yes }
			add = { desc = "ffpa_commonwealth_progress_increasing" value = 1 }
		}
		else = { subtract = { desc = "ffpa_commonwealth_progress_decreasing" value = 1 } }
	}

	default_green = yes
	start_value = 0
	min_value = 0
	max_value = 60
}
