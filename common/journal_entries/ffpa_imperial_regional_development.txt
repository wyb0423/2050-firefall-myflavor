# Regional construction journals shared by the restored Ottoman and Byzantine imperial paths.
# Each reward is a permanent physical state modifier and is deliberately scoped to the route it represents.

je_ffpa_via_egnatia = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	group = je_group_ffpa_imperial_development

	is_shown_when_inactive = {
		ffpa_imperial_regional_development_unlocked = yes
		NOT = { has_variable = ffpa_via_egnatia_complete }
	}

	possible = {
		ffpa_imperial_regional_development_unlocked = yes
		owns_entire_state_region = STATE_ALBANIA
		owns_entire_state_region = STATE_MACEDONIA
		owns_entire_state_region = STATE_WESTERN_THRACE
		owns_entire_state_region = STATE_EASTERN_THRACE
		NOT = { has_variable = ffpa_via_egnatia_complete }
	}

	complete = {
		any_scope_state = {
			state_region = s:STATE_ALBANIA
			market_access >= 0.90
			turmoil < 0.15
			any_scope_building = { is_building_type = building_railway level >= 3 }
			any_scope_building = { is_building_type = building_government_administration level >= 3 }
		}
		any_scope_state = {
			state_region = s:STATE_MACEDONIA
			market_access >= 0.90
			turmoil < 0.15
			any_scope_building = { is_building_type = building_railway level >= 5 }
			any_scope_building = { is_building_type = building_government_administration level >= 5 }
		}
		any_scope_state = {
			state_region = s:STATE_WESTERN_THRACE
			market_access >= 0.90
			turmoil < 0.15
			any_scope_building = { is_building_type = building_railway level >= 4 }
			any_scope_building = { is_building_type = building_government_administration level >= 4 }
		}
	}

	on_complete = {
		set_variable = ffpa_via_egnatia_complete
		every_scope_state = {
			limit = {
				NOT = { has_modifier = ffpa_region_via_egnatia }
				OR = {
					state_region = s:STATE_ALBANIA
					state_region = s:STATE_MACEDONIA
					state_region = s:STATE_WESTERN_THRACE
				}
			}
			add_modifier = ffpa_region_via_egnatia
		}
	}

	invalid = { NOT = { OR = { c:TUR ?= this c:BYZ ?= this } } }
	weight = 105
	should_be_pinned_by_default_uninvolved_or_context = no
}
