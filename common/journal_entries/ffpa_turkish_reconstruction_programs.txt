je_ffpa_tur_charter_of_the_porte = {
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	group = je_group_ffpa_turkish_reconstruction

	is_shown_when_inactive = {
		ffpa_tur_has_ottoman_state_project_v1 = yes
		has_variable = ffpa_tur_ottoman_commonwealth_complete
		NOT = { has_variable = ffpa_tur_porte_charter_complete_v1 }
	}

	possible = {
		ffpa_tur_has_ottoman_state_project_v1 = yes
		has_variable = ffpa_tur_ottoman_commonwealth_complete
		NOT = { has_variable = ffpa_tur_porte_charter_complete_v1 }
	}

	immediate = { ffpa_ensure_tur_route_event_chains_v1 = yes }

	complete = { ffpa_tur_porte_charter_completion_conditions_v1 = yes }

	on_complete = {
		set_variable = ffpa_tur_porte_charter_complete_v1
		ffpa_cleanup_tur_porte_chain_v1 = yes
		set_variable = ffpa_tur_porte_charter_capstone_scheduled_v1
		trigger_event = { id = ffpa_tur_flavor.13 days = 1 }
	}

	invalid = { NOT = { ffpa_tur_has_ottoman_state_project_v1 = yes } }
	on_invalid = { ffpa_cleanup_tur_porte_chain_v1 = yes }

	weight = 125
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_ffpa_tur_black_sea_lifeline = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_ffpa_turkish_reconstruction

	is_shown_when_inactive = {
		c:TUR ?= this
		has_variable = ffpa_tur_rebuild_anatolia_complete
		NOT = { has_variable = ffpa_tur_black_sea_lifeline_complete_v1 }
	}

	possible = {
		c:TUR ?= this
		has_variable = ffpa_tur_rebuild_anatolia_complete
		owns_entire_state_region = STATE_KASTAMONU
		owns_entire_state_region = STATE_TRABZON
		NOT = { has_variable = ffpa_tur_black_sea_lifeline_complete_v1 }
	}

	complete = {
		any_scope_state = {
			state_region = s:STATE_KASTAMONU
			market_access >= 0.85
			turmoil < 0.18
			any_scope_building = { is_building_type = building_port level >= 2 }
			any_scope_building = { is_building_type = building_railway level >= 3 }
			any_scope_building = { is_building_type = building_power_plant level >= 2 }
		}
		any_scope_state = {
			state_region = s:STATE_TRABZON
			market_access >= 0.85
			turmoil < 0.18
			any_scope_building = { is_building_type = building_port level >= 4 }
			any_scope_building = { is_building_type = building_railway level >= 3 }
			any_scope_building = { is_building_type = building_power_plant level >= 2 }
		}
	}

	on_complete = {
		set_variable = ffpa_tur_black_sea_lifeline_complete_v1
		every_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_KASTAMONU
					state_region = s:STATE_TRABZON
				}
				NOT = { has_modifier = ffpa_tur_black_sea_lifeline_state_v1 }
			}
			add_modifier = ffpa_tur_black_sea_lifeline_state_v1
		}
		ffpa_check_tur_post_reconstruction_events_v1 = yes
	}

	invalid = { NOT = { c:TUR ?= this } }
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = no
}
