# Optional Byzantine restoration campaigns. They are added only through the
# restoration-council decision, so their 45-55 year clocks begin when the player
# deliberately accepts a frontier. Claims are supplied by the formation module;
# these JEs only observe direct ownership and never alter war-goal definitions.

je_ffpa_byz_restore_balkans = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_ffpa_byzantine_restoration

	immediate = {
		add_modifier = { name = ffpa_byz_restoration_campaign_mobilization_v1 years = 8 }
	}

	complete = {
		owns_entire_state_region = STATE_SLOVENIA
		owns_entire_state_region = STATE_CROATIA
		owns_entire_state_region = STATE_SLAVONIA
		owns_entire_state_region = STATE_DALMATIA
		owns_entire_state_region = STATE_BOSNIA
		owns_entire_state_region = STATE_MONTENEGRO
		owns_entire_state_region = STATE_ALBANIA
		owns_entire_state_region = STATE_SKOPIA
		owns_entire_state_region = STATE_KOSOVO
		owns_entire_state_region = STATE_NORTHERN_THRACE
		owns_entire_state_region = STATE_BULGARIA
		owns_entire_state_region = STATE_EASTERN_SERBIA
		owns_entire_state_region = STATE_WESTERN_SERBIA
		owns_entire_state_region = STATE_PELOPONNESE
		owns_entire_state_region = STATE_IONIAN_ISLANDS
		owns_entire_state_region = STATE_ATTICA
		owns_entire_state_region = STATE_THESSALIA
		owns_entire_state_region = STATE_MACEDONIA
		owns_entire_state_region = STATE_WESTERN_THRACE
		owns_entire_state_region = STATE_CRETE
		owns_entire_state_region = STATE_WEST_AEGEAN_ISLANDS
		owns_entire_state_region = STATE_DOBRUDJA
		owns_entire_state_region = STATE_EASTERN_THRACE
	}

	on_complete = {
		set_variable = ffpa_byz_balkans_restored_complete_v1
		add_modifier = ffpa_byz_balkans_restored_v1
		trigger_event = { id = ffpa_flavor.40 popup = yes }
	}

	timeout = 16425 # 45 years
	on_timeout = {
		set_variable = { name = ffpa_byz_balkans_restoration_retry_cooldown_v1 years = 5 }
		add_modifier = { name = ffpa_byz_restoration_failure_v1 years = 8 }
		trigger_event = { id = ffpa_flavor.47 popup = yes }
	}

	invalid = { NOT = { country_definition = cd:BYZ } }
	on_invalid = {
		if = {
			limit = { has_modifier = ffpa_byz_restoration_campaign_mobilization_v1 }
			remove_modifier = ffpa_byz_restoration_campaign_mobilization_v1
		}
	}
	status_desc = { triggered_desc = { desc = je_ffpa_byz_restoration_campaign_status } }
	progressbar = no
	weight = 130
	should_be_pinned_by_default_uninvolved_or_context = yes
}
