ffpa_tur_ottoman_commonwealth_month_conditions = {
	c:TUR ?= this
	country_has_monarchy_law = yes
	OR = {
		has_law_or_variant = law_type:law_cultural_exclusion
		has_law_or_variant = law_type:law_multicultural
	}
	OR = {
		has_law_or_variant = law_type:law_freedom_of_conscience
		has_law_or_variant = law_type:law_total_separation
	}
	government_legitimacy >= 60
	is_at_war = no
	ffpa_has_no_insurrectionary_movement = yes

	calc_true_if = {
		amount >= 4
		OR = {
			cultural_acceptance_base = { target = cu:greek value >= acceptance_status_3 }
			cultural_acceptance_base = { target = cu:bulgarian value >= acceptance_status_3 }
			cultural_acceptance_base = { target = cu:serb value >= acceptance_status_3 }
			cultural_acceptance_base = { target = cu:romanian value >= acceptance_status_3 }

		OR = {
			cultural_acceptance_base = { target = cu:albanian value >= acceptance_status_3 }
			cultural_acceptance_base = { target = cu:bosniak value >= acceptance_status_3 }
		}
		cultural_acceptance_base = { target = cu:armenian value >= acceptance_status_3 }
		cultural_acceptance_base = { target = cu:kurdish value >= acceptance_status_3 }
		OR = {
			cultural_acceptance_base = { target = cu:arab value >= acceptance_status_3 }
			cultural_acceptance_base = { target = cu:egyptian value >= acceptance_status_3 }
		}
	}

	any_scope_state = {
		count >= 6
		OR = {
			state_region = s:STATE_EASTERN_THRACE
			state_region = s:STATE_HUDAVENDIGAR
			state_region = s:STATE_AYDIN

			state_region = s:STATE_ADANA

			state_region = s:STATE_ERZURUM
			state_region = s:STATE_DIYARBAKIR
			state_region = s:STATE_KARS
		}
		turmoil < 0.20
	}
}
