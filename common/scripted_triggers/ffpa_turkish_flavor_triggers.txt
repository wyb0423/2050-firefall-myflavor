# Country scope unless noted otherwise. These triggers are the stable query
# surface for the Turkish reconstruction state machine and its one-way links
# from completed regional construction projects.

ffpa_tur_has_ottoman_state_project_v1 = {
	c:TUR ?= this
	has_variable = ffpa_tur_state_project_v1
	var:ffpa_tur_state_project_v1 = 1
}

ffpa_tur_has_republican_state_project_v1 = {
	c:TUR ?= this
	has_variable = ffpa_tur_state_project_v1
	var:ffpa_tur_state_project_v1 = 2
}

ffpa_tur_has_directorate_state_project_v1 = {
	c:TUR ?= this
	has_variable = ffpa_tur_state_project_v1
	var:ffpa_tur_state_project_v1 = 3
}

# State scope. Firefall's final TUR formation list contains exactly these
# eleven state regions.
ffpa_is_tur_reconstruction_core_state_v1 = {
	OR = {
		state_region = s:STATE_EASTERN_THRACE
		state_region = s:STATE_HUDAVENDIGAR
		state_region = s:STATE_AYDIN
		state_region = s:STATE_ANKARA
		state_region = s:STATE_KONYA
		state_region = s:STATE_ADANA
		state_region = s:STATE_KASTAMONU
		state_region = s:STATE_TRABZON
		state_region = s:STATE_ERZURUM
		state_region = s:STATE_DIYARBAKIR
		state_region = s:STATE_KARS
	}
}

ffpa_tur_porte_charter_completion_conditions_v1 = {
	ffpa_tur_has_ottoman_state_project_v1 = yes
	has_variable = ffpa_tur_porte_waqf_event_fired_v1
	country_has_monarchy_law = yes
	government_legitimacy >= 60
	bureaucracy >= 0
	is_at_war = no
	ffpa_has_no_insurrectionary_movement = yes
	any_scope_state = {
		count >= 8
		ffpa_is_tur_reconstruction_core_state_v1 = yes
		market_access >= 0.75
		turmoil < 0.18
	}
}

ffpa_tur_second_foundation_completion_conditions_v1 = {
	ffpa_tur_has_republican_state_project_v1 = yes
	has_variable = ffpa_tur_republic_civil_military_event_fired_v1
	NOT = { country_has_monarchy_law = yes }
	country_has_voting_franchise = yes
	government_legitimacy >= 60
	bureaucracy >= 0
	literacy_rate >= 0.30
	is_at_war = no
	ffpa_has_no_insurrectionary_movement = yes
	any_scope_state = {
		count >= 8
		ffpa_is_tur_reconstruction_core_state_v1 = yes
		market_access >= 0.75
		turmoil < 0.18
	}
}

ffpa_tur_directorate_completion_conditions_v1 = {
	ffpa_tur_has_directorate_state_project_v1 = yes
	has_variable = ffpa_tur_directorate_technical_corps_event_fired_v1
	government_legitimacy >= 50
	bureaucracy >= 100
	literacy_rate >= 0.25
	ffpa_has_no_insurrectionary_movement = yes
	any_scope_state = {
		count >= 6
		ffpa_is_tur_reconstruction_core_state_v1 = yes
		market_access >= 0.75
		turmoil < 0.20
		any_scope_building = {
			is_building_type = building_railway
			level >= 2
		}
		any_scope_building = {
			is_building_type = building_government_administration
			level >= 2
		}
	}
}
