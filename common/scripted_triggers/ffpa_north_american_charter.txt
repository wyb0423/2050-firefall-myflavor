ffpa_usa_charter_callbacks_safe_v1 = {
 # A republic's native activation changes crown lands into puppets. Require ordinary resolution first.
 trigger_if = {
  limit = { ffpa_usa_charter_governance_changes_v1 = yes }
  custom_tooltip = {
   text = ffpa_usa_charter_crownlands_tt
   NOT = { any_subject_or_below = { is_subject_type = subject_type_crown_land is_direct_subject_of = ROOT } }
  }
  custom_tooltip = {
   text = ffpa_usa_charter_company_tt
   OR = { is_subject = yes NOT = { has_government_type = gov_chartered_company } }
  }
 }
}
