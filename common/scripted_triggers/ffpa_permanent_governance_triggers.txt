ffpa_byz_public_welfare_condition_count_v1 = {
	calc_true_if = {
		amount >= 2
		average_sol >= {
			value = var:ffpa_byz_public_sol_baseline_v1
			subtract = 1
		}
		AND = {
			var:ffpa_byz_public_incorporated_states_v1 > 0
			var:ffpa_byz_public_access_states_v1 >= {
				value = var:ffpa_byz_public_incorporated_states_v1

			}
		}
		var:ffpa_byz_public_famine_states_v1 <= {
			value = var:ffpa_byz_public_incorporated_states_v1
