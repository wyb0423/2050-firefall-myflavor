# Country scope. The register drifts through lightweight country and capital
# accessors; no building or state loop is needed.
ffpa_tur_register_monthly_balance_change_v1 = {
	value = 0
	if = {
		limit = { bureaucracy >= 0 }
		add = { value = 0.25 desc = ffpa_tur_register_monthly_balance_change_v1_factor_1 }
	}
	else = { subtract = { value = 0.50 desc = ffpa_tur_register_monthly_balance_change_v1_factor_2 } }
	if = {
		limit = { government_legitimacy >= 60 }
		add = { value = 0.25 desc = ffpa_tur_register_monthly_balance_change_v1_factor_3 }
	}
	else_if = {
		limit = { government_legitimacy < 40 }
		subtract = { value = 0.25 desc = ffpa_tur_register_monthly_balance_change_v1_factor_4 }
	}
	if = {
		limit = { in_default = yes }
		subtract = { value = 0.50 desc = ffpa_tur_register_monthly_balance_change_v1_factor_5 }
	}
	if = {
		limit = { capital = { market_access >= 0.85 } }
		add = { value = 0.25 desc = ffpa_tur_register_monthly_balance_change_v1_factor_6 }
	}
	else = { subtract = { value = 0.25 desc = ffpa_tur_register_monthly_balance_change_v1_factor_7 } }
	min = -1
	max = 0.75
}

# Country scope. Counts the four explicitly documented capital-supply
# anomalies without iterating states, pops, or buildings.
ffpa_tur_capital_supply_anomaly_count_v1 = {
	value = 0
	if = {
		limit = {
			market_capital.market = {
				mg:grain = { market_goods_pricier > 0.25 }
			}
		}
		add = { value = 1 desc = ffpa_tur_capital_supply_anomaly_count_v1_factor_1 }
	}
	if = {
		limit = { capital = { sg:transportation = { state_goods_pricier > 0.25 } } }
		add = { value = 1 desc = ffpa_tur_capital_supply_anomaly_count_v1_factor_2 }
	}
	if = {
		limit = { capital = { sg:electricity = { state_goods_pricier > 0.25 } } }
		add = { value = 1 desc = ffpa_tur_capital_supply_anomaly_count_v1_factor_3 }
	}
	if = {
		limit = { capital = { market_access < 0.85 } }
		add = { value = 1 desc = ffpa_tur_capital_supply_anomaly_count_v1_factor_4 }
	}
}

# Country scope. Stage reductions are encoded at the source so a completed
# reform never needs a second corrective pass.
ffpa_byz_dynatoi_monthly_pressure_gain_v1 = {
	value = 0
	if = {
		limit = { var:ffpa_byz_military_reform_stage_v1 < 1 }
		add = { value = 0.25 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_1 }
	}
	if = {
		limit = { ig:ig_landowners = { is_powerful = yes } }
		if = {
			limit = { var:ffpa_byz_military_reform_stage_v1 >= 3 }
			add = { value = 0.25 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_2 }
		}
		else = { add = { value = 0.50 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_3 } }
	}
	if = {
		limit = { has_law_or_variant = law_type:law_serfdom }
		if = {
			limit = { var:ffpa_byz_military_reform_stage_v1 >= 2 }
			add = { value = 0.50 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_4 }
		}
		else = { add = { value = 1.00 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_5 } }
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_tenant_farmers }
		if = {
			limit = { var:ffpa_byz_military_reform_stage_v1 >= 2 }
			add = { value = 0.25 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_6 }
		}
		else = { add = { value = 0.50 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_7 } }
	}
	if = {
		limit = {
			var:ffpa_byz_military_reform_stage_v1 < 1
			has_law_or_variant = law_type:law_peasant_levies
		}
		add = { value = 0.50 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_8 }
	}
	if = {
		limit = {
			var:ffpa_byz_military_reform_stage_v1 < 3
			OR = {
				has_modifier = ffpa_byz_western_provincial_tax_compact
				AND = {
					has_variable = ffpa_byz_western_cadastre_choice_v2
					var:ffpa_byz_western_cadastre_choice_v2 = 2
				}
			}
		}
		add = { value = 0.25 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_9 }
	}
	if = {
		limit = {
			has_variable = ffpa_byz_western_cadastre_choice_v2
			var:ffpa_byz_western_cadastre_choice_v2 = 1
		}
		subtract = { value = 0.25 desc = ffpa_byz_dynatoi_monthly_pressure_gain_v1_factor_10 }
	}
	min = 0
}

# Country scope. Same 0/1/2/3/4-anomaly brackets as the existing monthly pulse.

 value = 0
 if = { limit = { ffpa_tur_capital_supply_anomaly_count_v1 = 0 } add = { value = 3 desc = ffpa_tur_capital_anomaly_bracket_0 } }
 else_if = { limit = { ffpa_tur_capital_supply_anomaly_count_v1 = 1 } add = { value = 1 desc = ffpa_tur_capital_anomaly_bracket_1 } }

 else_if = { limit = { ffpa_tur_capital_supply_anomaly_count_v1 = 3 } add = { value = -5 desc = ffpa_tur_capital_anomaly_bracket_3 } }
 else_if = { limit = { ffpa_tur_capital_supply_anomaly_count_v1 = 4 } add = { value = -8 desc = ffpa_tur_capital_anomaly_bracket_4 } }
 if = { limit = { ffpa_tur_capital_has_famine_v1 = yes } subtract = { value = 10 desc = ffpa_tur_capital_famine_factor } }
}
