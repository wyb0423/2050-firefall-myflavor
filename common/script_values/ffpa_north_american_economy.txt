 value = 0
 every_scope_state = { limit = { ffpa_na_is_mainland_state_v1 = yes } add = state_population }
}
ffpa_usa_market_coverage_v1 = {
 value = 0
 if = {
  limit = { ffpa_usa_mainland_population_v1 > 0 }
  every_scope_state = { limit = { ffpa_na_is_mainland_state_v1 = yes market_access >= 0.95 } add = state_population }
  divide = ffpa_usa_mainland_population_v1
  multiply = 100
 }
}
