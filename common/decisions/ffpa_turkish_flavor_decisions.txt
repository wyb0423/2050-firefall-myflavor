# Old saves whose ten-year pragmatic reconstruction modifier has already
# expired cannot be migrated without guessing. This decision opens a one-time
# explicit recovery event and disappears as soon as a route is recorded.
ffpa_tur_confirm_reconstruction_legacy = {
	is_shown = {
		c:TUR ?= this
		has_variable = ffpa_tur_rebuild_anatolia_complete
		has_variable = ffpa_tur_state_project_repair_needed_v1
		NOT = { has_variable = ffpa_tur_state_project_v1 }
	}

	possible = {
		NOT = { has_variable = ffpa_tur_state_project_repair_event_scheduled_v1 }
	}

	when_taken = {
		set_variable = ffpa_tur_state_project_repair_event_scheduled_v1
		trigger_event = { id = ffpa_tur_flavor.1 popup = yes }
	}

	ai_chance = { value = 100 }
}
