# Visible manual entry for the vanilla Eastern Roman title proposal. The
# Greek Nationalism journal remains its automatic scheduler while active;
# this decision also keeps the political choice reachable after that journal
# has completed in an existing save.
ffpa_request_eastern_roman_title = {
	is_shown = {
		c:GRE ?= this
		has_variable = embrace_megali_idea_var
		NOT = { has_variable = embrace_ambitious_agenda_var }
		NOT = { has_variable = byzantium_event_fired }
	}

	possible = {
		country_has_monarchy_law = yes
		custom_tooltip = {
			text = ffpa_eastern_roman_title_ionian_tt
			any_scope_state = {
				state_region = s:STATE_IONIAN_ISLANDS
			}
		}
		government_legitimacy >= 75
	}

	when_taken = {
		trigger_event = { id = ffpa_flavor.8 popup = yes }
	}

	ai_chance = {
		value = 10
	}
}

# Reusable player-facing entry to the Byzantine restoration campaign tree.
# The event presents only branches whose prerequisites and retry cooldowns allow
# them; the decision itself enforces peace and the shared two-campaign cap.
ffpa_byz_convene_restoration_council = {
	is_shown = {
		c:BYZ ?= this
		ffpa_byz_has_available_restoration_campaign = yes
	}

	possible = {
		is_at_war = no
		custom_tooltip = {
			text = ffpa_byz_restoration_campaign_slot_available_tt
			ffpa_byz_restoration_campaign_slot_available = yes
		}
	}

	when_taken = {
		trigger_event = { id = ffpa_flavor.39 popup = yes }
	}

	ai_chance = {
		value = 10
	}
}
