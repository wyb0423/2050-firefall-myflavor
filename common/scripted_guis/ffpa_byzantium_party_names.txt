# BYZ party-name localization adapters.
#
# Party names are defined inside the global party database. Replacing those
# definitions for tag-only flavor would duplicate thousands of vanilla lines.
# The base game already uses scripted-GUI localization for Taikunate party
# names, so these adapters keep the vanilla generic name for every non-BYZ
# country and return a dedicated name only when the localization root is BYZ.

ffpa_byz_agrarian_party_name_sgui = {
	scope = country
	effect = {
		if = {
			limit = { this = c:BYZ }
			custom_tooltip_no_bullet = ffpa_party_byz_agrarian
		}
		else = { custom_tooltip_no_bullet = ffpa_party_agrarian_fallback }
	}
}
