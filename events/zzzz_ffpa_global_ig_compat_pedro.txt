				any_country = {
					has_journal_entry = je_portugal_united_kingdom

						binds = ROOT

							has_type = amend_succession
						}
					}
				}
			}
			random_country = {
				limit = {
					has_journal_entry = je_portugal_united_kingdom

							has_type = amend_succession
						}
					}
				}
				trigger_event = { id = lusofonia.4 popup = yes }
			}
		}
		if = {
			limit = {
				NOT = { has_law = law_type:law_landed_voting }
			}
			activate_law = law_type:law_landed_voting
		}

	}

	option = {
		name = pedro.8.a
		default_option = yes
		hidden_effect = {
			ig:ig_landowners ?= {
				ffpa_global_ig_remove_hierarchic = yes
				add_ideology = ideology_oligarchic

			}
		}
		if = {
			limit = {
				ig:ig_armed_forces ?= {
					is_in_government = no
					ig_counts_as_marginal = no
				}
			}
			ig:ig_armed_forces ?= {
				add_ruling_interest_group = yes
			}
		}

			limit = {
				ig:ig_armed_forces ?= {
					ig_counts_as_marginal = no
				}

			}
		}

			ig:ig_landowners ?= {

			name = abolishing_monarchy_var
			value = 0
		}

	}
