ffpa_tur_flavor.1 = {
	type = country_event
	placement = ROOT
	title = ffpa_tur_flavor.1.t
	desc = ffpa_tur_flavor.1.d
	flavor = ffpa_tur_flavor.1.f
	event_image = { video = "unspecific_signed_contract" }
	icon = "gfx/interface/icons/timed_modifier_icons/modifier_documents_positive.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		country_definition = cd:TUR
		has_variable = ffpa_tur_state_project_repair_event_scheduled_v1
		NOT = { has_variable = ffpa_tur_state_project_v1 }
	}

	immediate = { remove_variable = ffpa_tur_state_project_repair_event_scheduled_v1 }

	# Migration records the historical route only. It must not replay the finite
	# or permanent reward that the original reconstruction event already handled.
	option = {
		name = ffpa_tur_flavor.1.a
		ffpa_set_tur_state_project_ottoman_v1 = yes
		ffpa_ensure_tur_flavor_v1 = yes
		ai_chance = {
			factor = 1
			modifier = {
				trigger = { country_has_monarchy_law = yes }
				add = 1
			}
		}
	}

	option = {
		name = ffpa_tur_flavor.1.b
		default_option = yes
		ffpa_set_tur_state_project_republic_v1 = yes
		ffpa_ensure_tur_flavor_v1 = yes
		ai_chance = {
			factor = 1
			modifier = {
				trigger = { NOT = { country_has_monarchy_law = yes } }
				add = 1
			}
		}
	}

	option = {
		name = ffpa_tur_flavor.1.c
		ffpa_set_tur_state_project_directorate_v1 = yes
		ffpa_ensure_tur_flavor_v1 = yes
		ai_chance = {
			factor = 1
			modifier = {
				trigger = { has_law_or_variant = law_type:law_technocracy }
				add = 1
			}
		}
	}
}
