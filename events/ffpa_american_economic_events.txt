ffpa_usa_flavor.10 = {
 type = country_event
 placement = ROOT
 title = ffpa_usa_flavor.10.t

 flavor = ffpa_usa_flavor.10.f
 event_image = { video = "unspecific_signed_contract" }
 icon = "gfx/interface/icons/event_icons/event_trade.dds"

 duration = 3
 trigger = { ffpa_usa_economy_identity_v1 = yes var:ffpa_usa_economy_pending_v1 = 10 }
 immediate = {
  save_scope_value_as = { name = ffpa_usa_economy_event_serial value = var:ffpa_usa_economy_serial_v1 }
 }
 option = {
  name = ffpa_usa_flavor.10.1
  default_option = yes
  if = {
   limit = { ffpa_usa_economy_event_valid_v1 = { EVENT = 10 } }
   set_variable = { name = ffpa_usa_market_dispute_v1 value = 1 }
   set_variable = { name = ffpa_usa_market_grant_weekly_v1 value = ffpa_usa_economy_weekly_grant_v1 years = 2 }
   add_modifier = { name = ffpa_usa_market_grant_v1 multiplier = var:ffpa_usa_market_grant_weekly_v1 years = 2 }
   add_modifier = { name = ffpa_usa_market_linkage_v1 years = 2 }
   ffpa_usa_economy_finish_event_v1 = yes
  }
  ai_chance = { factor = 10 }
