namespace = formation

# Full override of vanilla Victoria 3 1.13.* formation.3.
# Difference from upstream: preserve the notification and prestige flow, but
# replace the broad region_balkans/region_near_east claim filters with the
# adapter-owned Justinianic whitelist.
formation.3 = {
	type = country_event
	placement = ROOT

	title = ffpa_formation_byzantium.t
	desc = ffpa_formation_byzantium.d
	flavor = ffpa_formation_byzantium.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	trigger = {
		country_definition = cd:BYZ
		NOT = { has_global_variable = has_formed_byz }
	}

	immediate = {
		set_global_variable = has_formed_byz
		save_scope_as = formation_country
		every_country = {
			limit = {
				has_diplomatic_relevance = ROOT
			}
			post_notification = unification_notification_byz
		}
	}

	option = {
		name = ffpa_formation_byzantium.a
		default_option = yes
		add_modifier = {
			name = unification_prestige
			days = very_long_modifier_time
		}
		ffpa_grant_byzantine_justinianic_claims_v2 = yes
		set_variable = ffpa_byzantine_justinianic_claims_v1
		set_variable = ffpa_byzantine_justinianic_claims_v2
		ai_chance = {
			factor = 1
		}
	}
}
